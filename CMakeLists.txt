cmake_minimum_required(VERSION 3.16)
project(simple-port-scanner VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(simple-port-scanner src/main.cpp src/scanner.cpp)
if(MSVC)
  target_compile_options(simple-port-scanner PRIVATE /W4)
  target_link_libraries(simple-port-scanner PRIVATE ws2_32)
else()
  target_compile_options(simple-port-scanner PRIVATE -Wall -Wextra -Wpedantic)
  find_package(Threads REQUIRED)
  target_link_libraries(simple-port-scanner PRIVATE Threads::Threads)
endif()
include(CTest)
find_package(Python3 COMPONENTS Interpreter)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME integration_tests
    COMMAND ${CMAKE_COMMAND} -E env SCANNER_BIN=$<TARGET_FILE:simple-port-scanner>
      ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
  set_tests_properties(integration_tests PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
